import { SlashCommandBuilder, MessageFlags } from 'discord.js';
import { successEmbed } from '../../utils/embeds.js';

export default {
    data: new SlashCommandBuilder()
        .setName('gmm')
        .setDescription('GMM commands')
        .addSubcommand(subcommand =>
            subcommand
                .setName('countdown')
                .setDescription('Show the countdown until 20 June 2027 at 00:00 CEST')
        ),

    async execute(interaction) {
        // Zorg ervoor dat dit daadwerkelijk een /gmm countdown command is
        if (
            !interaction.isChatInputCommand() ||
            interaction.commandName !== 'gmm' ||
            interaction.options.getSubcommand() !== 'countdown'
        ) {
            return;
        }

        await interaction.deferReply({
            flags: MessageFlags.Ephemeral
        });

        try {
            const now = Date.now();

            // 20 juni 2027 om 00:00 CEST
            // CEST = UTC+2, dus in UTC is dit 19 juni 2027 om 22:00
            const target = Date.UTC(
                2027,
                5,  // juni (0 = januari)
                19,
                22,
                0,
                0
            );

            const difference = target - now;

            if (difference <= 0) {
                const embed = successEmbed(
                    '🎉 GMM Countdown',
                    '**20 juni 2027 om 00:00 CEST is bereikt!** 🎉'
                );

                await interaction.editReply({
                    embeds: [embed]
                });

                return;
            }

            const totalSeconds = Math.floor(difference / 1000);

            const days = Math.floor(totalSeconds / 86400);
            const hours = Math.floor((totalSeconds % 86400) / 3600);
            const minutes = Math.floor((totalSeconds % 3600) / 60);
            const seconds = totalSeconds % 60;

            // Unix timestamp voor Discord timestamps
            const unixTimestamp = Math.floor(target / 1000);

            const embed = successEmbed(
                '🎪 GMM Countdown',
                [
                    '# ⏳ Nog te gaan',
                    '',
                    `**${days} dagen**`,
                    `**${hours} uur**`,
                    `**${minutes} minuten**`,
                    `**${seconds} seconden**`,
                    '',
                    '📅 **20 juni 2027**',
                    '🕛 **00:00 CEST**',
                    '',
                    `🗓️ <t:${unixTimestamp}:F>`,
                    `⏱️ <t:${unixTimestamp}:R>`
                ].join('\n')
            );

            await interaction.editReply({
                embeds: [embed]
            });
        } catch (error) {
            console.error('Failed to generate GMM countdown:', error);

            const errorMessage = {
                content: '❌ Er ging iets mis bij het genereren van de GMM countdown.',
                embeds: []
            };

            // Als de interaction al gedeferd is, moet editReply worden gebruikt
            await interaction.editReply(errorMessage);
        }
    }
};
